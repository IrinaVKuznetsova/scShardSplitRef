1 Overview

This vignette shows how to prepare a barley reference genome compatible with Cell Ranger ARC using scShardSplitRef.

Cell Ranger ARC cannot handle FASTA sequences longer than 2^29 bp (536,870,912 bp). Large chromosomes must therefore be split before running cellranger-arc mkref.

This vignette is written as a complete case study because reference preparation is often species- and repository-specific. In this example:

  • nuclear FASTA and GTF files are taken from Ensembl Plants release 63;
  • mitochondrial and plastid FASTA/GTF records are taken from Ensembl Plants release 48;
  • centromere coordinates are taken from GrainGenes.

The generic scShardSplitRef workflow starts once we have prepared:

  1. a combined FASTA;
  2. a combined GTF;
  3. a BED-like table with chromosome name, centromere position, and chromosome size.

3 Case study-specific preprocessing

3.1 Download files

hv_nucl_ep_rel_63 <- "https://ftp.ensemblgenomes.ebi.ac.uk/pub/plants/release-63/fasta/hordeum_vulgare/dna/Hordeum_vulgare.MorexV3_pseudomolecules_assembly.dna.toplevel.fa.gz"

hv_ep_rel_63_gtf <- "https://ftp.ensemblgenomes.ebi.ac.uk/pub/plants/release-63/gtf/hordeum_vulgare/Hordeum_vulgare.MorexV3_pseudomolecules_assembly.63.gtf.gz"

hv_mt_ep_rel_48 <- "https://ftp.ensemblgenomes.ebi.ac.uk/pub/plants/release-48/fasta/hordeum_vulgare/dna/Hordeum_vulgare.IBSC_v2.dna.chromosome.Mt.fa.gz"

hv_pt_ep_rel_48 <- "https://ftp.ensemblgenomes.ebi.ac.uk/pub/plants/release-48/fasta/hordeum_vulgare/dna/Hordeum_vulgare.IBSC_v2.dna.chromosome.Pt.fa.gz"

hv_ep_rel_48_gtf <- "https://ftp.ensemblgenomes.ebi.ac.uk/pub/plants/release-48/gtf/hordeum_vulgare/Hordeum_vulgare.IBSC_v2.48.gtf.gz"

hv_grain_genes_centromere_data <- "https://graingenes.org/GG3/sites/default/files/data_downloads/barley-morex3.zip"

cmd <- paste(
  "wget -P", shQuote(data_dir),
  shQuote(hv_nucl_ep_rel_63),
  shQuote(hv_ep_rel_63_gtf),
  shQuote(hv_mt_ep_rel_48),
  shQuote(hv_pt_ep_rel_48),
  shQuote(hv_ep_rel_48_gtf),
  shQuote(hv_grain_genes_centromere_data)
)

system(cmd)

Define local file paths.

nuclear_fa <- file.path(data_dir, "Hordeum_vulgare.MorexV3_pseudomolecules_assembly.dna.toplevel.fa.gz")
mt_fa <- file.path(data_dir, "Hordeum_vulgare.IBSC_v2.dna.chromosome.Mt.fa.gz")
pt_fa <- file.path(data_dir, "Hordeum_vulgare.IBSC_v2.dna.chromosome.Pt.fa.gz")

morex_gtf <- file.path(data_dir, "Hordeum_vulgare.MorexV3_pseudomolecules_assembly.63.gtf.gz")
organelle_gtf <- file.path(data_dir, "Hordeum_vulgare.IBSC_v2.48.gtf.gz")

centromere_zip <- file.path(data_dir, "barley-morex3.zip")

3.2 Combine FASTA files

The nuclear genome is written first, followed by mitochondrial and plastid sequences.

genome_fasta <- file.path(data_dir, "hv_combined.fa")

cmd <- paste(
  "zcat",
  shQuote(nuclear_fa),
  shQuote(mt_fa),
  shQuote(pt_fa),
  ">",
  shQuote(genome_fasta)
)

system(cmd)

stopifnot(file.exists(genome_fasta))

Index the combined FASTA and extract sequence sizes.

fa <- Rsamtools::FaFile(genome_fasta)

if (!file.exists(paste0(genome_fasta, ".fai"))) {

3.3 Prepare centromere coordinates

cmd <- paste(
  "unzip -o",
  shQuote(centromere_zip),
  "-d",
  shQuote(data_dir)
)

system(cmd)

hv_centromere <- Sys.glob(
  file.path(data_dir, "*", "*", "*", "MorexV3_centromere_positions.tsv")
)

stopifnot(length(hv_centromere) == 1)

centromere_bed <- file.path(data_dir, "hv_combined_centromeres.bed")

readr::read_delim(
  hv_centromere,
  col_names = c("chr", "centromere_pos"),
  show_col_types = FALSE
) |>
  mutate(chr = stringr::str_remove(chr, "^chr")) |>
  right_join(chr_sizes, by = "chr") |>
  mutate(centromere_pos = tidyr::replace_na(centromere_pos, 0L)) |>
  select(chr, centromere_pos, size_bp) |>
  readr::write_delim(
    file = centromere_bed,
    delim = "\t",
    col_names = FALSE
  )

stopifnot(file.exists(centromere_bed))

3.4 Combine GTF files

The MorexV3 release 63 GTF is used as the main annotation. Mitochondrial and plastid records are appended from release 48.

gtf_path <- file.path(data_dir, "hv_combined.gtf")

# The release 48 GTF contains annotations for both nuclear and organellar sequences. Only mitochondrial and plastid annotations are retained and appended to the release 63 nuclear annotation.
cmd <- sprintf(
  "(zcat %s; zcat %s | awk '$1==\"Mt\" || $1==\"Pt\"') > %s",
  shQuote(morex_gtf),
  shQuote(organelle_gtf),
  shQuote(gtf_path)
)

system(cmd)

stopifnot(file.exists(gtf_path))

Check that Mt and Pt records are present.

cmd <- sprintf(
  "awk '$1==\"Mt\" || $1==\"Pt\" {print $1}' %s | sort -u",
  shQuote(gtf_path)
)

system(cmd)

8 Split FASTA using bedtools

cmd <- paste(
  "bedtools getfasta",
  "-fi",
  shQuote(genome_fasta),
  "-bed",
  shQuote(split_bed_path),
  "| sed '/^>/s/:/-/g'",
  ">",
  shQuote(out_fasta)
)

system(cmd)

stopifnot(file.exists(out_fasta))

config_path <- file.path(output_dir, "03_cellranger_arc_barley.config")

12 Build the reference with cellranger

# Build the reference
cellranger_arc <- "/usr/local/bin/cellranger-arc"

mkref_log <- file.path(output_dir, "mkref.log")

exit_status <- system2(
   cellranger_arc,
  args = c(
    "mkref",
    paste0("--config=", config_path),
    "--nthreads=12"
  ),
  stdout = mkref_log,
  stderr = mkref_log
)

if (exit_status != 0) {
  stop("cellranger-arc mkref failed. Check log: ", mkref_log)
}
